<script>
async function carregarTudo() {
  try {
    const feedUrl = "/feeds/posts/default?alt=json&max-results=300";
    const res = await fetch(feedUrl);
    const data = await res.json();

    if (!data.feed || !data.feed.entry) return;

    const entries = data.feed.entry;

    const posts = [];
    const statusPosts = [];
    let ultimaImagem = null;
    let ultimoLink = null;
    let dataUltimaImagem = null;

    entries.forEach(entry => {
      const tags = entry.category ? entry.category.map(c => c.term.toLowerCase()) : [];
      const link = entry.link.find(l => l.rel === "alternate")?.href || "#";
      const dataPub = new Date(entry.published.$t);
      const titulo = entry.title.$t;

      if (tags.includes("notas semanais")) {
        statusPosts.push({ titulo, link, data: dataPub });
      }

      if (!tags.some(t => [".txt","fotografia","now","leituras","status","notas semanais",'música'].includes(t))) {
        posts.push({ titulo, link, data: dataPub });
      }

      const content = entry.content?.$t || "";
      const temp = document.createElement("div");
      temp.innerHTML = content;

      const imgs = temp.querySelectorAll('img[data-type="fotolog"]');

      if (imgs.length) {
        if (!dataUltimaImagem || dataPub > dataUltimaImagem) {
          ultimaImagem = imgs[imgs.length - 1];
          ultimoLink = link;
          dataUltimaImagem = dataPub;
        }
      }
    });

    posts.sort((a,b)=>b.data-a.data);
    statusPosts.sort((a,b)=>b.data-a.data);

    document.getElementById("ultimos-posts").innerHTML =
      `<div class="lista-posts">` +
      posts.slice(0,8).map(p => `<div class="item"><a href="${p.link}">${p.titulo}</a></div>`).join("") +
      `</div>`;

    document.getElementById("posts-status").innerHTML =
      statusPosts.map(p => `<div class="status-item"><a href="${p.link}">${p.titulo}</a></div>`).join("");

    if (ultimaImagem) {
      document.getElementById("fotologSingle").innerHTML =
        `<a href="${ultimoLink}">
          <img src="${ultimaImagem.src.replace(/s\\d+/, "s600")}">
        </a>`;
    }

  // Música
const musicaRes = await fetch(`/feeds/posts/default/-/música?alt=json&max-results=1`);
const musicaData = await musicaRes.json();

if (musicaData.feed && musicaData.feed.entry) {
  const post = musicaData.feed.entry[0];
  const title = post.title.$t;
  const parts = title.split(",");
  const album = parts[0]?.trim() || title;
  const artist = parts[1]?.trim() || "";
  const link = post.link.find(l => l.rel === "alternate")?.href || "#";

  let img = "";
  if (post.media$thumbnail) {
    img = post.media$thumbnail.url.replace(/\/s\d+(-c)?\//, "/s0/");
  }

  document.getElementById("album-grid").innerHTML = `
    <img src="${img}">
    <a class="album" href="${link}">
      <span class="album-title">${album}</span>
      <span class="album-artist">${artist}</span>
    </a>
  `;
}

    // Leituras
    const leituraRes = await fetch("/feeds/posts/default/-/leituras?alt=json&max-results=4");
    const leituraData = await leituraRes.json();

    if (leituraData.feed && leituraData.feed.entry) {
      const lista = leituraData.feed.entry.map(item => {
        const titulo = item.title.$t;
        
        const parts = titulo.split(",");
        const livro = parts[0]?.trim() || titulo;
        const autor = parts[1]?.trim() || "";
        
        const link = item.link.find(l => l.rel === "alternate")?.href || "#";
        const conteudo = item.content?.$t || "";
        const imgMatch = conteudo.match(/<img[^>]+src="([^">]+)"/i);
        const imagem = imgMatch ? imgMatch[1] : "";

        return `
          <div class="card">
            <a href="${link}" class="card-link">
              <div class="card-image-wrapper">
                ${imagem ? `<img src="${imagem}">` : `<div class="no-image">Sem imagem</div>`}
              </div>
              <div class="card-info">
                <div class="resenha-livro">${livro}</div>
                <div class="resenha-autor">${autor}</div>
              </div>
            </a>
          </div>
        `;
      }).join("");

      document.getElementById("leituras-lista").innerHTML =
        `<div class="leituras-gallery">${lista}</div>`;
    }

  } catch(e) {
    console.error(e);
  }
}

carregarTudo();
</script>

<script>
function loadStatus(json) {
  const container = document.getElementById("status-feeds");
  const entries = json.feed.entry;

  if (!entries) {
    container.innerHTML = "Nenhum status encontrado.";
    return;
  }

  entries.forEach(function(entry) {

    let content = entry.content ? entry.content.$t : entry.summary.$t;

    // Remove quebras de linha
    content = content.replace(/(\r\n|\n|\r)/gm, " ");

    // Link automático para @usuario (ajuste a URL base)
    content = content.replace(/@([a-zA-Z0-9_]+)/g, 
      '<a href="https://seusite.com/$1" target="_blank">@$1</a>'
    );

    // Link do post
    const postLink = entry.link.find(l => l.rel === "alternate").href;

    // Formatar data: dd . mm . yy, hh:mm
    const d = new Date(entry.published.$t);
    const day = String(d.getDate()).padStart(2, "0");
    const month = String(d.getMonth() + 1).padStart(2, "0");
    const year = String(d.getFullYear()).slice(-2);
    const hours = String(d.getHours()).padStart(2, "0");
    const minutes = String(d.getMinutes()).padStart(2, "0");
    const date = `${day} . ${month} . ${year} / ${hours}:${minutes} — `;

    const post = document.createElement("div");
    post.className = "status-post";

    post.innerHTML = `
      <div class="row">
  <span class="date"> ${date}</span> <span class="content"> ${content} </span>
</div>
      
    `;

    container.appendChild(post);
  });
}
</script>

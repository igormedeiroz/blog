<script>
const circles = document.querySelectorAll(".circulo");

let positions = [
  { x: 100, y: 100, dx: 2, dy: 2 },
  { x: 300, y: 200, dx: -2, dy: 1.5 },
   { x: 700, y: 200, dx: -2, dy: 3 }
];

function moveCircles() {
  const width = window.innerWidth;
  const height = window.innerHeight;

  circles.forEach((circle, i) => {
    let p = positions[i];

    p.x += p.dx;
    p.y += p.dy;

    if (p.x <= 0 || p.x + circle.offsetWidth >= width) p.dx *= -1;
    if (p.y <= 0 || p.y + circle.offsetHeight >= height) p.dy *= -1;

    circle.style.left = p.x + "px";
    circle.style.top = p.y + "px";
  });

  requestAnimationFrame(moveCircles);
}

moveCircles();
</script>
